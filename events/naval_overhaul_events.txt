# Event 1: Naval Doctrine & Arms Race Debate
naval_overhaul.1 = {
	type = country_event
	placement = root
	title = naval_overhaul.1.t
	desc = naval_overhaul.1.d
	flavor = naval_overhaul.1.f

	duration = 3

	event_image = {
		video = "unspecific_steam_ship"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_military.dds"

	option = {
		name = naval_overhaul.1.a
		default_option = yes
		if = {
			limit = { has_modifier = modifier_naval_arms_race_momentum }
			remove_modifier = modifier_naval_arms_race_momentum
		}
		add_modifier = {
			name = modifier_naval_arms_race_focus
			days = 1095
		}
	}

	option = {
		name = naval_overhaul.1.b
		if = {
			limit = { has_modifier = modifier_naval_arms_race_momentum }
			remove_modifier = modifier_naval_arms_race_momentum
		}
		add_modifier = {
			name = modifier_naval_construction_boom
			days = 1095
		}
	}

	option = {
		name = naval_overhaul.1.c
		if = {
			limit = { has_modifier = modifier_naval_arms_race_momentum }
			remove_modifier = modifier_naval_arms_race_momentum
		}
		add_modifier = {
			name = modifier_naval_victory_morale
			days = 1095
		}
	}
}

# Event 3: Sail-to-Ironclad Transition Friction
naval_overhaul.3 = {
	type = country_event
	placement = root
	title = naval_overhaul.3.t
	desc = naval_overhaul.3.d
	flavor = naval_overhaul.3.f

	duration = 3

	event_image = {
		video = "unspecific_steam_ship"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_military.dds"

	trigger = {
		has_technology_researched = ironclad_tech
		navy_size >= 15
		NOT = { has_variable = naval_overhaul_ironclad_transition_done }
	}

	immediate = {
		set_variable = naval_overhaul_ironclad_transition_done
	}

	option = {
		name = naval_overhaul.3.a
		default_option = yes
		add_modifier = {
			name = modifier_naval_ironclad_modernization_drive
			days = 1095
		}
		ig:ig_industrialists ?= {
			add_modifier = {
				name = modifier_naval_ironclad_modernization_drive
				days = 730
			}
		}
	}

	option = {
		name = naval_overhaul.3.b
		add_modifier = {
			name = modifier_naval_ironclad_transition_strain
			days = 730
		}
		add_loyalists = {
			value = small_radicals
			pop_type = officers
		}
	}
}
