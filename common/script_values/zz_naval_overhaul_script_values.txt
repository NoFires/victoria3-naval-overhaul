# Precision REPLACE for naval_battle_size only (preserves land_battle_size & other mods' command_values)
REPLACE:naval_battle_size = {
	value = 0
	if = {
		limit = {
			exists = scope:military_formation
			scope:military_formation = { is_fleet = yes }
		}
		value = {
			add = scope:military_formation.num_ships_not_in_battle
			desc = "NAVAL_BATTLE_SIZE_ADMIRAL_SHARE"
		}
	}
	else_if = {
		limit = {
			exists = scope:military_formation
		}
		value = {
			add = scope:military_formation.num_assigned_supply_ships
			desc = "NAVAL_BATTLE_SIZE_ADMIRAL_SHARE"
		}
	}
	else = {
		value = {
			add = scope:enemy_military_formation.num_ships
			desc = "NAVAL_BATTLE_SIZE_ADMIRAL_SHARE"
		}
	}

	save_temporary_value_as = pre_max_amount_cap

	if = {
		limit = {
			exists = scope:own_commander
			scope:pre_max_amount_cap > {
				value = 0
				scope:own_commander = {
					add = this.command_limit_num_units
				}
				multiply = 0.5
			}
		}
		max = {
			add = {
				scope:own_commander = {
					add = this.command_limit_num_units
				}
			}
			multiply = 0.5
			desc = "[concept_command_limit]"
		}
		min = define:NMilitary|MILITARY_FORMATION_COMMAND_LIMIT_NO_COMMANDER
	}
	else_if = {
		limit = {
			NOT = { exists = scope:own_commander }
		}
		max = define:NMilitary|MILITARY_FORMATION_COMMAND_LIMIT_NO_COMMANDER
	}

	save_temporary_value_as = post_command_limit_base_amount_cap

	# Custom Formula: ((Own Detection * Enemy Visibility) / (Enemy Detection * Own Visibility)) * Enemy Ships * 0.25
	if = {
		limit = {
			exists = scope:military_formation
			exists = scope:enemy_military_formation
			scope:military_formation = { is_fleet = yes }
			scope:enemy_military_formation = { is_fleet = yes }
		}
		value = scope:post_command_limit_base_amount_cap
		add = {
			add = {
				scope:military_formation = {
					add = this.detection
				}
			}
			multiply = scope:enemy_military_formation.visibility

			divide = {
				value = 0
				add = {
					scope:enemy_military_formation = {
						add = this.detection
					}
				}
				min = 1

				multiply = {
					value = scope:military_formation.visibility
					min = 1
				}
			}

			min = 0.35
			max = 2.5

			multiply = scope:enemy_military_formation.num_ships_not_in_battle
			multiply = 0.25
			desc = "NAVAL_BATTLE_SIZE_RECON_ADVANTAGE"
		}
	}
	else_if = {
		limit = {
			exists = scope:military_formation
			exists = scope:enemy_military_formation
		}
		value = scope:post_command_limit_base_amount_cap
	}
	else_if = {
		limit = {
			exists = scope:military_formation
			NOT = { exists = scope:enemy_military_formation }
		}
		value = scope:post_command_limit_base_amount_cap
		add = scope:military_formation.num_ships_not_in_battle
	}
	else = {
		value = scope:post_command_limit_base_amount_cap
		add = scope:enemy_military_formation.num_ships_not_in_battle
	}

	if = {
		limit = {
			exists = scope:military_formation
			scope:military_formation = { is_fleet = yes }
		}
		max = scope:military_formation.num_ships_not_in_battle
		if = {
			limit = { exists = scope:own_commander }
			max = {
				value = 0
				scope:own_commander = {
					add = this.command_limit_num_units
				}
				min = define:NMilitary|MILITARY_FORMATION_COMMAND_LIMIT_NO_COMMANDER
				desc = "[concept_command_limit]"
			}
		}
		else = {
			max = define:NMilitary|MILITARY_FORMATION_COMMAND_LIMIT_NO_COMMANDER
		}
	}
}

# Budget- & intent-aware scaling for expensive/high-tier Overhaul ship modifications
ship_mod_budget_high_mult = {
	value = 1.0
	if = {
		limit = { in_default = yes }
		multiply = 0.15
	}
	else_if = {
		limit = { scaled_debt >= 0.30 }
		multiply = 0.35
	}
	else_if = {
		limit = { scaled_debt >= 0.15 }
		multiply = 0.65
	}

	if = {
		limit = { military_wage_level = very_low }
		multiply = 0.25
	}
	else_if = {
		limit = { military_wage_level = low }
		multiply = 0.55
	}
	else_if = {
		limit = { military_wage_level = high }
		multiply = 1.20
	}
	else_if = {
		limit = { military_wage_level = very_high }
		multiply = 1.40
	}

	if = {
		limit = { is_at_war = no }
		multiply = 0.90
	}
	else = {
		multiply = 1.15
	}

	if = {
		limit = { country_rank < rank_value:major_power }
		multiply = 0.78
	}
	else_if = {
		limit = { country_rank < rank_value:great_power }
		multiply = 0.90
	}

	if = {
		limit = {
			OR = {
				has_strategy = ai_strategy_armed_isolationism
				has_strategy = ai_strategy_conservative_agenda
			}
		}
		multiply = 0.85
	}
	min = 0.35
}

# Moderate scaling for mid-tier Overhaul ship modifications
ship_mod_budget_mid_mult = {
	value = 1.0
	if = {
		limit = { in_default = yes }
		multiply = 0.45
	}
	else_if = {
		limit = { scaled_debt >= 0.30 }
		multiply = 0.70
	}

	if = {
		limit = { military_wage_level = very_low }
		multiply = 0.55
	}
	else_if = {
		limit = { military_wage_level = low }
		multiply = 0.85
	}
	else_if = {
		limit = {
			OR = {
				military_wage_level = high
				military_wage_level = very_high
			}
		}
		multiply = 1.15
	}

	if = {
		limit = { is_at_war = yes }
		multiply = 1.15
	}
	min = 0.25
}

# Cost-conscious scaling favoring economical/standard modules when austerity or low military wages apply
ship_mod_budget_low_mult = {
	value = 1.0
	if = {
		limit = {
			OR = {
				in_default = yes
				military_wage_level = very_low
			}
		}
		multiply = 1.75
	}
	else_if = {
		limit = {
			OR = {
				scaled_debt >= 0.25
				military_wage_level = low
			}
		}
		multiply = 1.40
	}
	else_if = {
		limit = {
			military_wage_level = very_high
			is_at_war = yes
		}
		multiply = 0.75
	}

	if = {
		limit = { country_rank < rank_value:major_power }
		multiply = 1.25
	}
	if = {
		limit = { is_at_war = no }
		multiply = 1.15
	}
}
