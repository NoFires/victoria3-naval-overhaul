on_game_started = {
	on_actions = {
		naval_overhaul_on_game_started
	}
}

on_ship_constructed = {
	on_actions = {
		naval_overhaul_on_ship_constructed
	}
}

on_naval_battle_won = {
	on_actions = {
		naval_overhaul_on_naval_battle_won
	}
}

on_flagship_battle_won = {
	on_actions = {
		naval_overhaul_on_flagship_battle_won
	}
}

on_monthly_pulse_country = {
	on_actions = {
		naval_overhaul_on_monthly_pulse_country
	}
	random_events = {
		chance_to_happen = 8
		100 = 0
		5 = naval_overhaul.2
		5 = naval_overhaul.3
		5 = naval_overhaul.4
		5 = naval_overhaul.5
		5 = naval_overhaul.6
	}
}

naval_overhaul_on_game_started = {
	effect = {
		set_global_variable = naval_overhaul_active
		set_global_variable = nte_naval_overhaul_detected
	}
}

naval_overhaul_on_ship_constructed = {
	effect = {
		if = {
			limit = {
				exists = scope:ship
				scope:ship = {
					OR = {
						is_ship_type = ship_type:ship_type_early_ironclad
						is_ship_type = ship_type:ship_type_modern_ironclad
						is_ship_type = ship_type:ship_type_pre_dreadnought
						is_ship_type = ship_type:ship_type_dreadnought
						is_ship_type = ship_type:ship_type_super_dreadnought
						is_ship_type = ship_type:ship_type_aircraft_carrier
					}
				}
				NOT = { has_variable = naval_overhaul_boom_cd }
				NOT = { has_modifier = modifier_naval_construction_boom }
				NOR = {
					has_variable = nte_country_holiday_accepted
					has_variable = nte_global_over_limit_flag
					has_variable = nte_bilateral_over_limit_flag
				}
			}
			set_variable = {
				name = naval_overhaul_boom_cd
				days = 365
			}
			add_modifier = {
				name = modifier_naval_construction_boom
				days = 90
			}
		}
	}
}

naval_overhaul_on_naval_battle_won = {
	effect = {
		if = {
			limit = {
				NOT = { has_modifier = modifier_naval_victory_morale }
			}
			add_modifier = {
				name = modifier_naval_victory_morale
				days = 90
			}
		}
	}
}

naval_overhaul_on_flagship_battle_won = {
	effect = {
		if = {
			limit = {
				NOT = { has_variable = naval_overhaul_flagship_triumph_cd }
			}
			set_variable = {
				name = naval_overhaul_flagship_triumph_cd
				days = 1825
			}
			trigger_event = {
				id = naval_overhaul.7
				popup = yes
			}
		}
	}
}

naval_overhaul_on_monthly_pulse_country = {
	effect = {
		if = {
			limit = { NOT = { has_global_variable = naval_overhaul_active } }
			set_global_variable = naval_overhaul_active
			set_global_variable = nte_naval_overhaul_detected
		}
		if = {
			limit = { navy_size >= 300 }
			if = { limit = { has_modifier = modifier_naval_hegemony_tier_1 } remove_modifier = modifier_naval_hegemony_tier_1 }
			if = { limit = { has_modifier = modifier_naval_hegemony_tier_2 } remove_modifier = modifier_naval_hegemony_tier_2 }
			if = {
				limit = { NOT = { has_modifier = modifier_naval_hegemony_tier_3 } }
				add_modifier = { name = modifier_naval_hegemony_tier_3 }
			}
		}
		else_if = {
			limit = { navy_size >= 150 }
			if = { limit = { has_modifier = modifier_naval_hegemony_tier_1 } remove_modifier = modifier_naval_hegemony_tier_1 }
			if = { limit = { has_modifier = modifier_naval_hegemony_tier_3 } remove_modifier = modifier_naval_hegemony_tier_3 }
			if = {
				limit = { NOT = { has_modifier = modifier_naval_hegemony_tier_2 } }
				add_modifier = { name = modifier_naval_hegemony_tier_2 }
			}
		}
		else_if = {
			limit = { navy_size >= 50 }
			if = { limit = { has_modifier = modifier_naval_hegemony_tier_2 } remove_modifier = modifier_naval_hegemony_tier_2 }
			if = { limit = { has_modifier = modifier_naval_hegemony_tier_3 } remove_modifier = modifier_naval_hegemony_tier_3 }
			if = {
				limit = { NOT = { has_modifier = modifier_naval_hegemony_tier_1 } }
				add_modifier = { name = modifier_naval_hegemony_tier_1 }
			}
		}
		else = {
			if = { limit = { has_modifier = modifier_naval_hegemony_tier_1 } remove_modifier = modifier_naval_hegemony_tier_1 }
			if = { limit = { has_modifier = modifier_naval_hegemony_tier_2 } remove_modifier = modifier_naval_hegemony_tier_2 }
			if = { limit = { has_modifier = modifier_naval_hegemony_tier_3 } remove_modifier = modifier_naval_hegemony_tier_3 }
		}
	}
}
