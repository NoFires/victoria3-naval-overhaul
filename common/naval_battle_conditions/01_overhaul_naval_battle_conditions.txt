naval_condition_night_action = {
	icon = "gfx/interface/icons/sea_condition_icons/fog.dds"

	possible = {
		always = yes
	}

	weight = {
		value = 1
		if = {
			limit = {
				scope:region = {
					has_harvest_condition = fog
				}
			}
			multiply = 3
		}
		if = {
			limit = {
				OR = {
					scope:attacker = {
						any_scope_ship = {
							OR = {
								is_ship_type = ship_type:ship_type_torpedo_boat
								is_ship_type = ship_type:ship_type_destroyer
							}
						}
					}
					scope:defender = {
						any_scope_ship = {
							OR = {
								is_ship_type = ship_type:ship_type_torpedo_boat
								is_ship_type = ship_type:ship_type_destroyer
							}
						}
					}
				}
			}
			add = 1
		}
	}

	period = {
		value = 0
	}

	modifier = {
		ship_accuracy_mult = -0.30
		ship_screening_mult = -0.15
		ship_readiness_gain_mult = -0.10
	}
}
